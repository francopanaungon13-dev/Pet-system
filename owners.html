const email = localStorage.getItem("currentUserEmail"); function getUsers() { return JSON.parse(localStorage.getItem("happyPawsUsers") || "[]"); } document.getElementById("petForm").addEventListener("submit", function(event) { event.preventDefault(); const users = getUsers(); const index = users.findIndex(user => user.email === email); if (index === -1) { alert("Account not found. Please register or log in."); window.location.href = "index.html"; return; } const pet = { id: Date.now().toString(), name: document.getElementById("petName").value.trim(), type: document.getElementById("petType").value, breed: document.getElementById("breed").value.trim(), gender: document.getElementById("gender").value, dob: document.getElementById("dob").value, color: document.getElementById("color").value.trim() }; users[index].pets = users[index].pets || []; users[index].pets.push(pet); try { localStorage.setItem("happyPawsUsers", JSON.stringify(users)); alert("Pet information saved successfully!"); window.location.href = "dashboard.html"; } catch (error) { alert("Unable to save pet information."); } });
