const email = localStorage.getItem("currentUserEmail"); function getUsers() { return JSON.parse(localStorage.getItem("happyPawsUsers") || "[]"); } function getAccount() { return getUsers().find(user => user.email === email); } let account = getAccount(); if (!account) { alert("Please register or log in first."); window.location.href = "index.html"; } else { document.querySelector(".subtitle").textContent = "Owner Information - " + account.firstName + " " + account.lastName; const owner = account.owner || {}; document.getElementById("address").value = owner.address || ""; document.getElementById("dob").value = owner.dob || ""; document.getElementById("gender").value = owner.gender || ""; document.getElementById("idType").value = owner.idType || ""; document.getElementById("idNumber").value = owner.idNumber || ""; document.getElementById("relationship").value = owner.relationship || ""; } document.getElementById("ownerForm").addEventListener("submit", function(e) { e.preventDefault(); const users = getUsers(); const index = users.findIndex(user => user.email === email); if (index === -1) { alert("Account not found. Please register again."); window.location.href = "index.html"; return; } users[index].owner = { ownerName: users[index].firstName + " " + users[index].lastName, address: document.getElementById("address").value.trim(), dob: document.getElementById("dob").value, gender: document.getElementById("gender").value, idType: document.getElementById("idType").value, idNumber: document.getElementById("idNumber").value.trim(), relationship: document.getElementById("relationship").value }; try { localStorage.setItem("happyPawsUsers", JSON.stringify(users)); window.location.href = "pet.html"; } catch (error) { alert("Unable to save owner information."); } });
